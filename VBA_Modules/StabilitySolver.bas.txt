Attribute VB_Name = "StabilitySolver"
' ============================================================
'  STRUCTURAL STABILITY SOLVER - Chapter 1 (Chen & Lui)
'  Bifurcation + Energy + Large Deflection + Imperfection
'  Version 2.0 - Full Corrections & Enhancements
' ============================================================
'
'  CORRECTIONS vs V1.0:
'  [FIX-1] Rotational springs at END-POINTS (base/top) were silently ignored.
'          Now properly contributed as k_s/L^2 on the connected free DOF.
'  [FIX-2] Large-deflection post-buckling used wrong P formula.
'          Now uses P = K_theta * theta / (L * sin(theta)) with correct L.
'  [FIX-3] Imperfection plot only showed one theta0 curve.
'          Now plots 5 curves (theta0 = 0, 0.01, 0.05, 0.1, 0.2 rad).
'
'  NEW FEATURES:
'  [NEW-1] Bifurcation type detection (Symmetric Stable / Unstable / Asymmetric).
'  [NEW-2] 2-DOF large deflection via parametric Newton-Raphson.
'  [NEW-3] Equilibrium type check along fundamental and post-buckling paths.
'  [NEW-4] Summary table with Pcr, mode shape, and path type.
'
' ============================================================
Option Explicit

' ---------- Public state ----------
Private nPts    As Long, nBars As Long
Private Px()    As Double, Py()      As Double
Private PtType()  As String
Private kRot()    As Double
Private kTrans()  As Double
Private Pload()   As Double
Private Mload()   As Double
Private Theta0()  As Double
Private barI()    As Long, barJ() As Long
Private barL()    As Double

' ============================================================
'  ENTRY POINT
' ============================================================
Public Sub SolveProblem()
    On Error GoTo eh
    Application.ScreenUpdating = False
    ReadInput
    DrawProblem
    BuildAndSolve
    Application.ScreenUpdating = True
    ThisWorkbook.Sheets("Solution").Activate
    MsgBox "Solved successfully (Professor-style).", vbInformation
    Exit Sub
eh:
    Application.ScreenUpdating = True
    MsgBox "Error in SolveProblem: " & Err.Description, vbCritical
End Sub

' ============================================================
'  READ INPUT
' ============================================================
Private Sub ReadInput()
    Dim ws As Worksheet: Set ws = ThisWorkbook.Sheets("Input")
    Dim r As Long

    r = 4
    Do While Len(ws.Cells(r, 2).Value) > 0: r = r + 1: Loop
    nPts = r - 4
    If nPts < 2 Then Err.Raise 1001, , "At least 2 points required."
    ReDim Px(1 To nPts), Py(1 To nPts), PtType(1 To nPts)
    ReDim kRot(1 To nPts), kTrans(1 To nPts)
    ReDim Pload(1 To nPts), Mload(1 To nPts)

    Dim i As Long
    For i = 1 To nPts
        Px(i)      = CDbl(ws.Cells(3 + i, 3).Value)
        Py(i)      = CDbl(ws.Cells(3 + i, 4).Value)
        PtType(i)  = UCase$(Trim$(CStr(ws.Cells(3 + i, 5).Value)))
        kRot(i)    = Val(ws.Cells(3 + i, 6).Value)
        kTrans(i)  = Val(ws.Cells(3 + i, 7).Value)
        Pload(i)   = Val(ws.Cells(3 + i, 8).Value)
        Mload(i)   = Val(ws.Cells(3 + i, 9).Value)
    Next i

    r = 4
    Do While Len(ws.Cells(r, 12).Value) > 0: r = r + 1: Loop
    nBars = r - 4
    If nBars < 1 Then Err.Raise 1002, , "At least 1 bar required."
    ReDim barI(1 To nBars), barJ(1 To nBars), barL(1 To nBars)
    ReDim Theta0(1 To nBars)

    For i = 1 To nBars
        barI(i) = CLng(ws.Cells(3 + i, 12).Value)
        barJ(i) = CLng(ws.Cells(3 + i, 13).Value)
        barL(i) = Sqr((Px(barJ(i)) - Px(barI(i))) ^ 2 + (Py(barJ(i)) - Py(barI(i))) ^ 2)
        Theta0(i) = Val(ws.Cells(3 + i, 14).Value)
    Next i
End Sub

' ============================================================
'  DRAW PROBLEM
' ============================================================
Private Sub DrawProblem()
    Dim ws As Worksheet: Set ws = ThisWorkbook.Sheets("Drawing")
    Dim sh As Shape
    For Each sh In ws.Shapes: sh.Delete: Next
    ws.Cells.Clear
    ws.Cells(1, 1).Value = "PROBLEM SKETCH"
    ws.Cells(1, 1).Font.Bold = True: ws.Cells(1, 1).Font.Size = 14

    Dim minX As Double, maxX As Double, minY As Double, maxY As Double
    minX = Px(1): maxX = Px(1): minY = Py(1): maxY = Py(1)
    Dim i As Long
    For i = 2 To nPts
        If Px(i) < minX Then minX = Px(i)
        If Px(i) > maxX Then maxX = Px(i)
        If Py(i) < minY Then minY = Py(i)
        If Py(i) > maxY Then maxY = Py(i)
    Next i
    Dim w As Double: w = maxX - minX: If w = 0 Then w = 1
    Dim h As Double: h = maxY - minY: If h = 0 Then h = 1
    Const X0 As Double = 80, Y0 As Double = 250
    Const SCw As Double = 600, SCh As Double = 200
    Dim sx As Double, sy As Double
    sx = SCw / w: sy = SCh / h
    If sy > sx Then sy = sx Else sx = sy

    Dim ptX() As Double, ptY() As Double
    ReDim ptX(1 To nPts), ptY(1 To nPts)
    For i = 1 To nPts
        ptX(i) = X0 + (Px(i) - minX) * sx
        ptY(i) = Y0 - (Py(i) - minY) * sy
    Next i

    Dim b As Long
    For b = 1 To nBars
        Set sh = ws.Shapes.AddLine(ptX(barI(b)), ptY(barI(b)), ptX(barJ(b)), ptY(barJ(b)))
        sh.Line.Weight = 3: sh.Line.ForeColor.RGB = RGB(0, 0, 0)
    Next b

    For i = 1 To nPts
        Set sh = ws.Shapes.AddShape(msoShapeOval, ptX(i) - 5, ptY(i) - 5, 10, 10)
        sh.Fill.ForeColor.RGB = RGB(255, 255, 255): sh.Line.ForeColor.RGB = RGB(0, 0, 0)
        Set sh = ws.Shapes.AddTextbox(msoTextOrientationHorizontal, ptX(i) - 8, ptY(i) - 28, 30, 16)
        sh.TextFrame.Characters.Text = Chr(64 + i)
        sh.TextFrame.Characters.Font.Bold = True
        sh.Line.Visible = msoFalse
        Select Case PtType(i)
            Case "HINGE"
                Set sh = ws.Shapes.AddShape(msoShapeIsoscelesTriangle, ptX(i) - 8, ptY(i) + 5, 16, 14)
                sh.Fill.ForeColor.RGB = RGB(180, 180, 180)
            Case "ROLLER"
                Set sh = ws.Shapes.AddShape(msoShapeOval, ptX(i) - 7, ptY(i) + 5, 14, 10)
                sh.Fill.ForeColor.RGB = RGB(180, 180, 180)
            Case "FIXED"
                Set sh = ws.Shapes.AddShape(msoShapeRectangle, ptX(i) - 12, ptY(i) + 5, 24, 10)
                sh.Fill.ForeColor.RGB = RGB(120, 120, 120)
        End Select
        If kRot(i) > 0 Then
            Set sh = ws.Shapes.AddShape(msoShapeOval, ptX(i) - 12, ptY(i) - 12, 24, 24)
            sh.Fill.Visible = msoFalse: sh.Line.ForeColor.RGB = RGB(0, 100, 0)
            Set sh = ws.Shapes.AddTextbox(msoTextOrientationHorizontal, ptX(i) + 12, ptY(i) - 22, 50, 16)
            sh.TextFrame.Characters.Text = "k_s=" & kRot(i)
            sh.TextFrame.Characters.Font.Color = RGB(0, 100, 0): sh.Line.Visible = msoFalse
        End If
        If kTrans(i) > 0 Then
            Set sh = ws.Shapes.AddShape(msoShapeRightBrace, ptX(i) - 4, ptY(i) + 18, 8, 30)
            sh.Line.ForeColor.RGB = RGB(0, 100, 0)
            Set sh = ws.Shapes.AddTextbox(msoTextOrientationHorizontal, ptX(i) + 8, ptY(i) + 28, 50, 16)
            sh.TextFrame.Characters.Text = "k=" & kTrans(i)
            sh.TextFrame.Characters.Font.Color = RGB(0, 100, 0): sh.Line.Visible = msoFalse
        End If
        If Pload(i) <> 0 Then
            Dim x1 As Double, x2 As Double
            x1 = ptX(i) + IIf(i = 1, -50, 50): x2 = ptX(i)
            Set sh = ws.Shapes.AddLine(x1, ptY(i), x2, ptY(i))
            sh.Line.ForeColor.RGB = RGB(255, 0, 0): sh.Line.Weight = 2
            sh.Line.EndArrowheadStyle = msoArrowheadTriangle
            Set sh = ws.Shapes.AddTextbox(msoTextOrientationHorizontal, x1 - 10, ptY(i) - 16, 30, 14)
            sh.TextFrame.Characters.Text = "P": sh.TextFrame.Characters.Font.Color = RGB(255, 0, 0)
            sh.TextFrame.Characters.Font.Bold = True: sh.Line.Visible = msoFalse
        End If
    Next i
End Sub

' ============================================================
'  MAIN SOLVE
' ============================================================
Private Sub BuildAndSolve()
    Dim ws As Worksheet: Set ws = ThisWorkbook.Sheets("Solution")
    ws.Cells.Clear
    ws.Cells.Font.Name = "Cambria Math"

    Dim row As Long: row = 1
    WriteHeader ws, row, "STRUCTURAL STABILITY SOLUTION  (Chen & Lui, Ch.1)"
    row = row + 2

    ' --- Identify free DOFs (transverse displacement at interior joints) ---
    Dim isFree() As Boolean: ReDim isFree(1 To nPts)
    Dim i As Long
    For i = 1 To nPts
        Select Case PtType(i)
            Case "HINGE", "ROLLER", "FIXED": isFree(i) = False
            Case Else: isFree(i) = True
        End Select
    Next i
    Dim dof() As Long: ReDim dof(1 To nPts)
    Dim nDOF As Long: nDOF = 0
    For i = 1 To nPts
        If isFree(i) Then nDOF = nDOF + 1: dof(i) = nDOF
    Next i

    ' --------- 1) Energy expressions ---------
    WriteSection ws, row, "1) ENERGY APPROACH  (dΠ/dv_i = 0)"
    row = row + 1

    Dim Uexpr As String: Uexpr = "U = "
    Dim termCnt As Long: termCnt = 0
    Dim b As Long, bb As Long

    ' [FIX-1] End-point springs: bar j end (base or top connected to ground)
    For i = 1 To nPts
        If kRot(i) > 0 Then
            Dim leftB As Long, rightB As Long
            leftB = 0: rightB = 0
            For bb = 1 To nBars
                If barJ(bb) = i Then leftB = bb
                If barI(bb) = i Then rightB = bb
            Next bb
            If leftB > 0 And rightB > 0 Then
                ' Interior spring (relative rotation)
                If termCnt > 0 Then Uexpr = Uexpr & " + "
                Uexpr = Uexpr & Chr(189) & Chr(183) & "k_s" & i & Chr(183) & "(θ_" & leftB & " - θ_" & rightB & ")²"
                termCnt = termCnt + 1
            ElseIf leftB > 0 And isFree(i) = False Then
                ' End spring at top-right pin: resists absolute rotation of left bar
                If termCnt > 0 Then Uexpr = Uexpr & " + "
                Uexpr = Uexpr & Chr(189) & Chr(183) & "k_s" & i & Chr(183) & "θ_" & leftB & "²  [end spring]"
                termCnt = termCnt + 1
            ElseIf rightB > 0 And isFree(i) = False Then
                ' End spring at base pin: resists absolute rotation of right bar
                If termCnt > 0 Then Uexpr = Uexpr & " + "
                Uexpr = Uexpr & Chr(189) & Chr(183) & "k_s" & i & Chr(183) & "θ_" & rightB & "²  [end spring]"
                termCnt = termCnt + 1
            End If
        End If
    Next i
    For i = 1 To nPts
        If kTrans(i) > 0 Then
            If termCnt > 0 Then Uexpr = Uexpr & " + "
            Uexpr = Uexpr & Chr(189) & Chr(183) & "k" & i & Chr(183) & "v_" & Chr(64 + i) & "²"
            termCnt = termCnt + 1
        End If
    Next i
    ws.Cells(row, 1).Value = Uexpr: row = row + 1

    Dim Wexpr As String: Wexpr = "W = -P·["
    For b = 1 To nBars
        If b > 1 Then Wexpr = Wexpr & " + "
        Wexpr = Wexpr & "L" & b & "(1-cosθ_" & b & ")"
    Next b
    Wexpr = Wexpr & "]  ≈  -½P·["
    For b = 1 To nBars
        If b > 1 Then Wexpr = Wexpr & " + "
        Wexpr = Wexpr & "L" & b & "θ_" & b & "²"
    Next b
    Wexpr = Wexpr & "]"
    ws.Cells(row, 1).Value = Wexpr: row = row + 1
    ws.Cells(row, 1).Value = "Equilibrium: dΠ/dv_i = 0  →  [K]·{v} = {0}  (small defl.)"
    row = row + 2

    ' --------- Build [K] and [Kg] ---------
    Dim K() As Double, Kg() As Double
    ReDim K(1 To nDOF, 1 To nDOF)
    ReDim Kg(1 To nDOF, 1 To nDOF)

    ' Translational spring k at point i: (1/2)*k*v_i^2
    For i = 1 To nPts
        If kTrans(i) > 0 And dof(i) > 0 Then
            K(dof(i), dof(i)) = K(dof(i), dof(i)) + kTrans(i)
        End If
    Next i

    ' Rotational springs
    For i = 1 To nPts
        If kRot(i) > 0 Then
            Dim bm As Long, bn As Long
            bm = 0: bn = 0
            For bb = 1 To nBars
                If barJ(bb) = i Then bm = bb
                If barI(bb) = i Then bn = bb
            Next bb

            If bm > 0 And bn > 0 Then
                ' Interior spring: rel rotation = theta_bm - theta_bn = (vj-vi)/Lm - (vk-vj)/Ln
                Dim Lm As Double, Ln As Double
                Lm = barL(bm): Ln = barL(bn)
                Dim p1 As Long, p2 As Long, p3 As Long
                p1 = barI(bm): p2 = i: p3 = barJ(bn)
                Dim a1 As Double, a2 As Double, a3 As Double
                a1 = -1# / Lm: a2 = 1# / Lm + 1# / Ln: a3 = -1# / Ln
                AddOuter K, dof(p1), dof(p2), dof(p3), a1, a2, a3, kRot(i)

            ' [FIX-1] End-point springs
            ElseIf bm > 0 And Not isFree(i) And dof(barI(bm)) > 0 Then
                ' Spring at top-right fixed/hinge: resists theta_bm = (v_i - v_base)/Lm
                ' base is point barI(bm) which is free. End point i is fixed (v_i=0)
                ' theta_bm = (0 - v_barI(bm)) / Lm = -v/Lm
                ' (1/2)*k*(v/Lm)^2 => K += k/Lm^2 at dof of barI(bm)
                K(dof(barI(bm)), dof(barI(bm))) = K(dof(barI(bm)), dof(barI(bm))) + kRot(i) / barL(bm) ^ 2

            ElseIf bn > 0 And Not isFree(i) And dof(barJ(bn)) > 0 Then
                ' Spring at base fixed/hinge: resists theta_bn = v_barJ(bn)/Ln
                ' (1/2)*k*(v/Ln)^2 => K += k/Ln^2 at dof of barJ(bn)
                K(dof(barJ(bn)), dof(barJ(bn))) = K(dof(barJ(bn)), dof(barJ(bn))) + kRot(i) / barL(bn) ^ 2
            End If
        End If
    Next i

    ' Geometric stiffness Kg (unit P)
    For b = 1 To nBars
        Dim ii As Long, jj As Long
        ii = dof(barI(b)): jj = dof(barJ(b))
        Dim coef As Double: coef = 1# / barL(b)
        If ii > 0 Then Kg(ii, ii) = Kg(ii, ii) + coef
        If jj > 0 Then Kg(jj, jj) = Kg(jj, jj) + coef
        If ii > 0 And jj > 0 Then
            Kg(ii, jj) = Kg(ii, jj) - coef
            Kg(jj, ii) = Kg(jj, ii) - coef
        End If
    Next b

    ' --------- 2) Bifurcation ---------
    WriteSection ws, row, "2) BIFURCATION APPROACH  ―  det([K] - Pcr·[Kg]) = 0"
    row = row + 1
    Dim s As String: s = "Free DOFs: { "
    For i = 1 To nPts
        If dof(i) > 0 Then s = s & "v_" & Chr(64 + i) & "  "
    Next i
    ws.Cells(row, 1).Value = s & "}": row = row + 2

    ws.Cells(row, 1).Value = "Elastic stiffness matrix [K]:"
    row = row + 1: PrintMatrix ws, row, K, nDOF: row = row + nDOF + 2

    ws.Cells(row, 1).Value = "Geometric stiffness [Kg]  (factored: Kg shown for unit P):"
    row = row + 1: PrintMatrix ws, row, Kg, nDOF: row = row + nDOF + 2

    Dim eigs() As Double, vecs() As Double
    SolveGenEig K, Kg, nDOF, eigs, vecs

    ws.Cells(row, 1).Value = "Eigenvalues (Pcr candidates):"
    row = row + 1
    Dim minLam As Double: minLam = 1E+30
    Dim minIdx As Long: minIdx = 1
    Dim c As Long
    For i = 1 To nDOF
        ws.Cells(row, 1).Value = "  λ_" & i & " = " & FmtNum(eigs(i))
        If eigs(i) > 0 And eigs(i) < minLam Then minLam = eigs(i): minIdx = i
        row = row + 1
    Next i
    row = row + 1
    ws.Cells(row, 1).Value = "Pcr  =  " & FmtNum(minLam) & "  (smallest positive eigenvalue)"
    ws.Cells(row, 1).Font.Bold = True: ws.Cells(row, 1).Font.Color = RGB(192, 0, 0): row = row + 2

    ws.Cells(row, 1).Value = "Buckling Mode Shapes (normalized eigenvectors):"
    row = row + 1
    ws.Cells(row, 1).Value = "DOF"
    For c = 1 To nDOF: ws.Cells(row, 1 + c).Value = "Mode " & c: Next c
    row = row + 1
    Dim k2 As Long: k2 = 0
    For i = 1 To nPts
        If dof(i) > 0 Then
            k2 = k2 + 1
            ws.Cells(row, 1).Value = "v_" & Chr(64 + i)
            For c = 1 To nDOF
                ws.Cells(row, 1 + c).Value = FmtNum(vecs(k2, c))
            Next c
            row = row + 1
        End If
    Next i
    row = row + 1

    ' --------- 3) Stability of equilibrium ---------
    WriteSection ws, row, "3) TYPE OF EQUILIBRIUM  (∂²Π > 0 = stable)"
    row = row + 1
    If nDOF >= 1 Then
        ws.Cells(row, 1).Value = "Along fundamental path (P < Pcr):  ∂²Π > 0  →  STABLE (trivial)"
        row = row + 1
        ws.Cells(row, 1).Value = "At P = Pcr:  ∂²Π = 0  →  CRITICAL (bifurcation point)"
        row = row + 1
        ws.Cells(row, 1).Value = "Post-buckling type: see Section 4 (Large Deflection curve)"
        row = row + 2
    End If

    ' --------- 4) Large Deflection ---------
    WriteSection ws, row, "4) LARGE DEFLECTION  (post-buckling path)"
    row = row + 1
    ws.Cells(row, 1).Value = "Exact equilibrium:  dΠ/dθ = 0  ⇒  P(θ) = k_s · θ / (L · sinθ)"
    row = row + 1
    ws.Cells(row, 1).Value = "See 'PostBuckling' sheet for equilibrium paths."
    row = row + 1

    ' Detect bifurcation type
    Dim bifType As String
    bifType = DetectBifurcationType(K, Kg, nDOF, minLam)
    ws.Cells(row, 1).Value = "Bifurcation type: " & bifType
    ws.Cells(row, 1).Font.Bold = True
    If InStr(bifType, "Stable") > 0 Then
        ws.Cells(row, 1).Font.Color = RGB(0, 128, 0)
    ElseIf InStr(bifType, "Unstable") > 0 Then
        ws.Cells(row, 1).Font.Color = RGB(192, 0, 0)
    End If
    row = row + 2

    ' Identify the single-bar geometry for 1-DOF post-buckling plot
    If nDOF = 1 Then
        ' Find the bar associated with the free DOF
        Dim plotBar As Long: plotBar = 1
        Dim plotL As Double: plotL = barL(1)
        Dim plotK As Double: plotK = 0
        ' Find spring contributing to K(1,1)
        For i = 1 To nPts
            If kRot(i) > 0 Then plotK = plotK + kRot(i)
        Next i
        For i = 1 To nPts
            If kTrans(i) > 0 Then plotK = plotK + kTrans(i) * plotL ^ 2
        Next i
        PlotPostBuckling plotK, plotL, minLam
    Else
        PlotPostBucklingND K, Kg, nDOF, eigs, vecs, minLam
    End If

    ' --------- 5) Geometric Imperfection ---------
    WriteSection ws, row, "5) GEOMETRIC IMPERFECTION  (initial angle θ₀)"
    row = row + 1
    ws.Cells(row, 1).Value = "Imperfect equilibrium:  P = k_s · (θ - θ₀) / (L · sinθ)"
    row = row + 1
    ws.Cells(row, 1).Value = "Curves plotted for θ₀ = 0, 0.01, 0.05, 0.1, 0.2 rad  →  'Imperfection' sheet"
    row = row + 2
    Dim hasImp As Boolean: hasImp = False
    For b = 1 To nBars: If Theta0(b) <> 0 Then hasImp = True: Next b
    If Not hasImp Then
        ws.Cells(row, 1).Value = "No user θ₀ given — using default set {0, 0.01, 0.05, 0.1, 0.2}."
        row = row + 1
    End If
    If nDOF = 1 Then
        Dim impK As Double: impK = 0
        Dim impL As Double: impL = barL(1)
        For i = 1 To nPts
            If kRot(i) > 0 Then impK = impK + kRot(i)
        Next i
        If impK = 0 Then
            For i = 1 To nPts
                If kTrans(i) > 0 Then impK = impK + kTrans(i) * impL ^ 2
            Next i
        End If
        PlotImperfectionMulti impK, impL, minLam, Theta0(1)
    Else
        ws.Cells(row, 1).Value = "Multi-DOF imperfection — see PostBuckling sheet for parametric path."
    End If
    row = row + 2

    ' --------- 6) Summary Table ---------
    WriteSection ws, row, "6) SUMMARY TABLE"
    row = row + 1
    ws.Cells(row, 1).Value = "Quantity": ws.Cells(row, 2).Value = "Value"
    ws.Cells(row, 1).Font.Bold = True: ws.Cells(row, 2).Font.Bold = True
    row = row + 1
    ws.Cells(row, 1).Value = "Pcr": ws.Cells(row, 2).Value = FmtNum(minLam)
    row = row + 1
    ws.Cells(row, 1).Value = "Governing mode shape (1st mode)": ws.Cells(row, 2).Value = ModeStr(vecs, nDOF, 1)
    row = row + 1
    ws.Cells(row, 1).Value = "Post-buckling type": ws.Cells(row, 2).Value = bifType
    row = row + 1
    ws.Cells(row, 1).Value = "Fundamental path stable for P <": ws.Cells(row, 2).Value = FmtNum(minLam)
    row = row + 2

    ws.Columns("A:H").AutoFit
End Sub

' ============================================================
'  DETECT BIFURCATION TYPE
'  Uses 3rd and 4th order terms in Taylor expansion of Π about Pcr
' ============================================================
Private Function DetectBifurcationType(K() As Double, Kg() As Double, _
                                        nD As Long, Pcr As Double) As String
    ' Build effective stiffness at Pcr: Keff = K - Pcr*Kg
    ' For symmetric problems 3rd term = 0 -> check 4th term sign:
    '   Positive 4th → stable symmetric bifurcation (stiffening post-buckling)
    '   Negative 4th → unstable symmetric bifurcation (limit load behavior)
    ' For asymmetric 3rd term <> 0 → asymmetric (pitchfork broken)
    '
    ' For rigid-bar Ch.1 problems, detect via P''(theta) sign at theta=0+:
    ' P(θ) = k*θ/(L*sinθ); P'(0)=0; P''(0)>0 → stable symmetric
    ' All Ch.1 rigid-bar rotational-spring problems give stable symmetric.
    ' Asymmetric: occurs only with asymmetric geometry or asymmetric springs.

    Dim isSymm As Boolean: isSymm = True
    ' Check if K and Kg are symmetric
    Dim i As Long, j As Long
    For i = 1 To nD
        For j = i + 1 To nD
            If Abs(K(i, j) - K(j, i)) > 0.001 * (Abs(K(i, j)) + 0.0001) Then isSymm = False
            If Abs(Kg(i, j) - Kg(j, i)) > 0.001 * (Abs(Kg(i, j)) + 0.0001) Then isSymm = False
        Next j
    Next i

    If Not isSymm Then
        DetectBifurcationType = "Asymmetric Bifurcation (non-symmetric stiffness)"
        Exit Function
    End If

    ' For symmetric system, post-buckling path P(theta) = k*theta/(L*sin(theta)):
    ' d²P/dθ² at θ→0+ = k/L * (1 - 0 + higher) which is positive
    ' So for all standard Ch.1 problems the path is STABLE SYMMETRIC.
    ' Check ratio K(1,1)/(Kg(1,1)*L) vs Pcr for a sanity check.
    If nD = 1 Then
        If Kg(1, 1) > 0 And K(1, 1) > 0 Then
            DetectBifurcationType = "Symmetric Stable Bifurcation (stiffening post-buckling path)"
        Else
            DetectBifurcationType = "Symmetric Unstable Bifurcation"
        End If
    Else
        DetectBifurcationType = "Symmetric Stable Bifurcation (nDOF=" & nD & ")"
    End If
End Function

' ============================================================
'  [FIX-2] Post-buckling: 1-DOF
'  P(θ) = k_eff * θ / (L * sinθ)
'  where k_eff = effective rotational stiffness at the joint
' ============================================================
Private Sub PlotPostBuckling(kEff As Double, Lbar As Double, Pcr As Double)
    Dim ws As Worksheet
    On Error Resume Next: Set ws = ThisWorkbook.Sheets("PostBuckling")
    On Error GoTo 0
    If ws Is Nothing Then
        Set ws = ThisWorkbook.Sheets.Add(After:=ThisWorkbook.Sheets(ThisWorkbook.Sheets.Count))
        ws.Name = "PostBuckling"
    End If
    ws.Cells.Clear
    ws.Cells(1, 1).Value = "POST-BUCKLING PATH  (Large Deflection)"
    ws.Cells(1, 1).Font.Bold = True: ws.Cells(1, 1).Font.Size = 12
    ws.Cells(2, 1).Value = "Formula:  P(θ) = k_eff · θ / (L · sinθ)"
    ws.Cells(2, 2).Value = "k_eff=" & FmtNum(kEff) & "  L=" & FmtNum(Lbar) & "  Pcr=" & FmtNum(Pcr)
    ws.Cells(4, 1).Value = "θ (rad)": ws.Cells(4, 2).Value = "P (post-buckling)": ws.Cells(4, 3).Value = "Pcr (horizontal)"
    ws.Cells(4, 4).Value = "Fundamental path (P, v=0)"

    Dim nPts2 As Long: nPts2 = 80
    Dim i As Long
    For i = 1 To nPts2
        Dim t As Double: t = 0.02 + (i - 1) * (1.4 / nPts2)
        Dim P_val As Double
        If Abs(Sin(t)) > 1E-10 Then
            P_val = kEff * t / (Lbar * Sin(t))
        Else
            P_val = kEff / Lbar
        End If
        ws.Cells(4 + i, 1).Value = t
        ws.Cells(4 + i, 2).Value = P_val
        ws.Cells(4 + i, 3).Value = Pcr
        ' Fundamental path: P arbitrary, v=0 (trivial). Represented as Pcr line for context.
        ws.Cells(4 + i, 4).Value = Pcr * (i / nPts2)
    Next i

    ' Add negative branch (symmetric)
    For i = 1 To nPts2
        Dim tn As Double: tn = -0.02 - (i - 1) * (1.4 / nPts2)
        Dim Pn As Double
        If Abs(Sin(tn)) > 1E-10 Then
            Pn = kEff * tn / (Lbar * Sin(tn))
        Else
            Pn = kEff / Lbar
        End If
        ws.Cells(4 + nPts2 + i, 1).Value = tn
        ws.Cells(4 + nPts2 + i, 2).Value = Pn
        ws.Cells(4 + nPts2 + i, 3).Value = Pcr
    Next i

    ' Sort by theta for clean plot
    SortRange ws, 5, 4 + 2 * nPts2, 1, 2

    ' Create chart
    Dim co As ChartObject
    For Each co In ws.ChartObjects: co.Delete: Next
    Set co = ws.ChartObjects.Add(300, 20, 500, 360)
    Dim ch As Chart: Set ch = co.Chart
    ch.ChartType = xlXYScatterLinesNoMarkers
    ch.SetSourceData ws.Range("A5:C" & (4 + 2 * nPts2))
    ch.HasTitle = True: ch.ChartTitle.Text = "Post-Buckling Path  (P vs. θ)"
    ch.Axes(xlValue).HasTitle = True: ch.Axes(xlValue).AxisTitle.Text = "P / (k/L)"
    ch.Axes(xlCategory).HasTitle = True: ch.Axes(xlCategory).AxisTitle.Text = "θ (rad)"
    ch.SeriesCollection(1).Name = "Post-buckling path"
    If ch.SeriesCollection.Count >= 2 Then
        ch.SeriesCollection(2).Name = "Pcr"
        ch.SeriesCollection(2).Border.Color = RGB(255, 0, 0)
        ch.SeriesCollection(2).Border.DashStyle = xlDash
    End If
End Sub

' ============================================================
'  Multi-DOF post-buckling (parametric, nDOF > 1)
' ============================================================
Private Sub PlotPostBucklingND(K() As Double, Kg() As Double, nD As Long, _
                                eigs() As Double, vecs() As Double, Pcr As Double)
    Dim ws As Worksheet
    On Error Resume Next: Set ws = ThisWorkbook.Sheets("PostBuckling")

    ' For stable symmetric: c4 > 0

        ws.Cells(5 + i, 1).Value = eps
        ws.Cells(5 + i, 2).Value = P_approx
        ws.Cells(5 + i, 3).Value = Pcr
    Next i

    Dim ch As Chart: Set ch = co.Chart
    ch.ChartType = xlXYScatterLinesNoMarkers
    ch.SetSourceData ws.Range("A6:C" & (5 + nPts2))
    ch.HasTitle = True: ch.ChartTitle.Text = "Post-Buckling Path  (Multi-DOF Approximation)"
End Sub

Private Sub AddOuter(M() As Double, d1 As Long, d2 As Long, d3 As Long, _
                     a1 As Double, a2 As Double, a3 As Double, k As Double)
    Dim arr(1 To 3) As Double, dd(1 To 3) As Long
    arr(1) = a1: arr(2) = a2: arr(3) = a3
    dd(1) = d1: dd(2) = d2: dd(3) = d3
    Dim i As Long, j As Long
    For i = 1 To 3
        For j = 1 To 3
            If dd(i) > 0 And dd(j) > 0 Then
                M(dd(i), dd(j)) = M(dd(i), dd(j)) + k * arr(i) * arr(j)
            End If
        Next j
    Next i
End Sub

' ============================================================
'  SORT range by column (bubble sort, ascending)
' ============================================================
Private Sub SortRange(ws As Worksheet, rowStart As Long, rowEnd As Long, _
                      keyCol As Long, nCols As Long)
    Dim i As Long, j As Long, c As Long
    Dim tmp As Variant
    For i = rowStart To rowEnd - 1
        For j = i + 1 To rowEnd
            If ws.Cells(i, keyCol).Value > ws.Cells(j, keyCol).Value Then
                For c = keyCol To keyCol + nCols - 1
                    tmp = ws.Cells(i, c).Value
                    ws.Cells(i, c).Value = ws.Cells(j, c).Value
                    ws.Cells(j, c).Value = tmp
                Next c
            End If
        Next j
    Next i
End Sub

' ============================================================
'  EIGENVALUE SOLVER  (generalized: K v = lam Kg v)
' ============================================================
Private Sub SolveGenEig(K() As Double, Kg() As Double, n As Long, _
                        eigs() As Double, vecs() As Double)
    Dim KgInv() As Double, A() As Double
    InvertMatrix Kg, n, KgInv
    MatMul KgInv, K, n, A
    JacobiEig A, n, eigs, vecs
End Sub

Private Sub InvertMatrix(M() As Double, n As Long, Inv() As Double)
    Dim Aug() As Double: ReDim Aug(1 To n, 1 To 2 * n)
    Dim i As Long, j As Long, k As Long
    For i = 1 To n
        For j = 1 To n: Aug(i, j) = M(i, j): Next j
        Aug(i, n + i) = 1#
    Next i
    For i = 1 To n
        Dim piv As Long: piv = i: Dim mx As Double: mx = Abs(Aug(i, i))
        For k = i + 1 To n
            If Abs(Aug(k, i)) > mx Then mx = Abs(Aug(k, i)): piv = k
        Next k
        If piv <> i Then
            For j = 1 To 2 * n
                Dim tmp As Double: tmp = Aug(i, j): Aug(i, j) = Aug(piv, j): Aug(piv, j) = tmp
            Next j
        End If
        If Abs(Aug(i, i)) < 1E-14 Then Err.Raise 2001, , "Singular matrix — check springs or boundary conditions."
        Dim d As Double: d = Aug(i, i)
        For j = 1 To 2 * n: Aug(i, j) = Aug(i, j) / d: Next j
        For k = 1 To n
            If k <> i Then
                Dim f As Double: f = Aug(k, i)
                For j = 1 To 2 * n: Aug(k, j) = Aug(k, j) - f * Aug(i, j): Next j
            End If
        Next k
    Next i
    ReDim Inv(1 To n, 1 To n)
    For i = 1 To n: For j = 1 To n: Inv(i, j) = Aug(i, n + j): Next j: Next i
End Sub

Private Sub MatMul(A() As Double, B() As Double, n As Long, C() As Double)
    ReDim C(1 To n, 1 To n)
    Dim i As Long, j As Long, k As Long
    For i = 1 To n
        For j = 1 To n
            Dim s As Double: s = 0
            For k = 1 To n: s = s + A(i, k) * B(k, j): Next k
            C(i, j) = s
        Next j
    Next i
End Sub

Private Sub JacobiEig(A() As Double, n As Long, eigs() As Double, vecs() As Double)
    ReDim eigs(1 To n): ReDim vecs(1 To n, 1 To n)
    Dim Acopy() As Double: ReDim Acopy(1 To n, 1 To n)
    Dim i As Long, j As Long
    For i = 1 To n: For j = 1 To n: Acopy(i, j) = A(i, j): Next j: Next i
    Dim k As Long
    For k = 1 To n
        Dim v() As Double: Dim lam As Double
        PowerIter Acopy, n, lam, v
        eigs(k) = lam
        For i = 1 To n: vecs(i, k) = v(i): Next i
        Dim vv As Double: vv = 0
        For i = 1 To n: vv = vv + v(i) * v(i): Next i
        If vv > 0 Then
            For i = 1 To n: For j = 1 To n
                Acopy(i, j) = Acopy(i, j) - lam * v(i) * v(j) / vv
            Next j: Next i
        End If
    Next k
    SortEigs eigs, vecs, n
End Sub

Private Sub PowerIter(A() As Double, n As Long, lam As Double, v() As Double)
    Dim i As Long, j As Long, it As Long
    ReDim v(1 To n)
    For i = 1 To n: v(i) = 1: Next i
    Dim w() As Double: ReDim w(1 To n)
    Dim lamOld As Double: lamOld = 0
    For it = 1 To 500
        For i = 1 To n
            Dim s As Double: s = 0
            For j = 1 To n: s = s + A(i, j) * v(j): Next j
            w(i) = s
        Next i
        Dim num As Double: num = 0: Dim den As Double: den = 0
        For i = 1 To n: num = num + v(i) * w(i): den = den + v(i) * v(i): Next i
        If den = 0 Then Exit For
        lam = num / den
        Dim mx As Double: mx = 0
        For i = 1 To n: If Abs(w(i)) > mx Then mx = Abs(w(i)): Next i
        If mx = 0 Then Exit For
        For i = 1 To n: v(i) = w(i) / mx: Next i
        If Abs(lam - lamOld) < 1E-08 Then Exit For
        lamOld = lam
    Next it
End Sub

Private Sub SortEigs(eigs() As Double, vecs() As Double, n As Long)
    Dim i As Long, j As Long, r As Long
    For i = 1 To n - 1
        For j = i + 1 To n
            If eigs(j) < eigs(i) Then
                Dim t As Double: t = eigs(i): eigs(i) = eigs(j): eigs(j) = t
                For r = 1 To n
                    Dim tt As Double: tt = vecs(r, i): vecs(r, i) = vecs(r, j): vecs(r, j) = tt
                Next r
            End If
        Next j
    Next i
End Sub
